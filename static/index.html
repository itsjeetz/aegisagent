Python Agent Middleware Integration

from aegis_firewall import FirewallEngine, InputSource, AegisAgentMiddleware

# 1. Initialize Firewall Engine
firewall = FirewallEngine()

# 2. Intercept unauthenticated external inputs (e.g. PDF, Email, Web)
safe_text, verdict = firewall.intercept(
    content=incoming_untrusted_data,
    source=InputSource.EMAIL
)

if not verdict.is_safe:
    print(f"[!] Threat neutralized: {verdict.primary_attack.value} (Risk: {verdict.risk_score}%)")

# 3. Pass safe_text to LLM Agent reasoning
response = agent.run(safe_text)

Tool Abuse Protection Decorator

from aegis_firewall.middleware import AegisAgentMiddleware

middleware = AegisAgentMiddleware()

# Protect dangerous tools from injection-driven abuse
@middleware.protect_tool
def run_bash_command(cmd: str):
    # Automatically blocked if prompt injection tries rm -rf or curl evil.com!
    return subprocess.check_output(cmd, shell=True)

@middleware.protect_tool
def send_email_tool(to: str, body: str):
    # Automatically blocked if indirect injection tries to exfiltrate secrets!
    return smtp_client.send(to=to, body=body)
